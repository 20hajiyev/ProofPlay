cmake_minimum_required(VERSION 3.19)
project(CombatRacer LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(RACER_WITH_ENGINE "Build targets that link the pinned Wicked Engine fork" ON)

enable_testing()
add_subdirectory(third_party)
add_subdirectory(game/core)
add_subdirectory(game/audio)

if(RACER_WITH_ENGINE)
    # Engine patch queue (docs/ENGINE_BASELINE.md). Fail early instead of at link time.
    file(READ ${PROJECT_SOURCE_DIR}/engine/WickedEngine/WickedEngine/wiPhysics.h WICKED_PHYSICS_H)
    if(NOT WICKED_PHYSICS_H MATCHES "racer P-001")
        message(FATAL_ERROR "Engine patch P-001 missing. Run: git -C engine/WickedEngine apply ../patches/0001-expose-vehicle-constraint.patch")
    endif()
    file(READ ${PROJECT_SOURCE_DIR}/engine/WickedEngine/WickedEngine/wiPhysics_Jolt.cpp WICKED_PHYSICS_JOLT_CPP)
    if(NOT WICKED_PHYSICS_JOLT_CPP MATCHES "racer P-002")
        message(FATAL_ERROR "Engine patch P-002 missing. Run: git -C engine/WickedEngine apply ../patches/0002-deterministic-body-creation.patch")
    endif()
    file(READ ${PROJECT_SOURCE_DIR}/engine/WickedEngine/WickedEngine/wiRenderPath3D.h WICKED_RENDERPATH3D_H)
    if(NOT WICKED_RENDERPATH3D_H MATCHES "racer P-003")
        message(FATAL_ERROR "Engine patch P-003 missing. Run: git -C engine/WickedEngine apply ../patches/0003-toon-crease-ink.patch")
    endif()
    set(WICKED_EDITOR OFF CACHE BOOL "" FORCE)
    set(WICKED_TESTS OFF CACHE BOOL "" FORCE)
    set(WICKED_IMGUI_EXAMPLE OFF CACHE BOOL "" FORCE)
    set(WICKED_WINDOWS_TEMPLATE OFF CACHE BOOL "" FORCE)
    set(WICKED_LINUX_TEMPLATE OFF CACHE BOOL "" FORCE)
    set(WICKED_ENABLE_SYMLINKS OFF CACHE BOOL "" FORCE)
    add_subdirectory(engine/WickedEngine EXCLUDE_FROM_ALL)
    add_subdirectory(tools/import_check)
    add_subdirectory(game/runtime)
    add_subdirectory(game/sandbox)
    add_subdirectory(proofplay)
endif()

add_subdirectory(game/tests)
